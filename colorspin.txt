@name Color do a spinny
@inputs [P1,P2,P3,P4,P5,P6,P7,P8]:entity Activate Col1:vector Col2:vector
@outputs Active
@persist Active

if(first() | dupefinished()){
    if(Activate == 1){
        Active = 1
        timer("T1",1)
    }
    else{
        Active = 0
    }
}

if(~Activate){
    if(Activate == 1){
        if(Active == 0){
            Active = 1
            timer("T1",1)
        }
    }
    else{
        if(Active == 1){
            Active = 0
            stopAllTimers()
            P1:setColor(Col2)
            P2:setColor(Col2)
            P3:setColor(Col2)
            P4:setColor(Col2)
            P5:setColor(Col2)
            P6:setColor(Col2)
            P7:setColor(Col2)
            P8:setColor(Col2)
        }
    }
}

if(clk("T1")){  
    P1:setColor(Col1)
    P8:setColor(Col2)
    
    timer("T2",140)
}

if(clk("T2")){  
    P2:setColor(Col1)
    P1:setColor(Col2)
    timer("T3",140)
}

if(clk("T3")){  
    P3:setColor(Col1)
    P2:setColor(Col2)
    timer("T4",140)
}

if(clk("T4")){  
    P4:setColor(Col1)
    P3:setColor(Col2)
    timer("T5",140)
}

if(clk("T5")){  
    P5:setColor(Col1)
    P4:setColor(Col2)
    timer("T6",140)
}


if(clk("T6")){  
    P6:setColor(Col1)
    P5:setColor(Col2)
    timer("T7",140)
}

if(clk("T7")){  
    P7:setColor(Col1)
    P6:setColor(Col2)
    timer("T8",140)
}

if(clk("T8")){  
    P8:setColor(Col1)
    P7:setColor(Col2)
    timer("T1",140)
}
